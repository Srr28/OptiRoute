cmake_minimum_required(VERSION 3.16)
project(OptiRoute VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(optiroute src/optiroute.cpp)
add_executable(optiroute_phase2 src/phase2.cpp)
add_executable(optiroute_validate_large validator/dinic_validator.cpp)

if(MSVC)
  target_compile_options(optiroute PRIVATE /W4)
  target_compile_options(optiroute_phase2 PRIVATE /W4)
  target_compile_options(optiroute_validate_large PRIVATE /W4)
else()
  target_compile_options(optiroute PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(optiroute_phase2 PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(optiroute_validate_large PRIVATE -Wall -Wextra -Wpedantic)
endif()
